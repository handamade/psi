# Docs site built with Psi (D121) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace `apps/promo` with an Astro app, `apps/site`, that serves the existing homepage as real HTML and adds `/docs/` — guides, 35 generated component pages and six token pages — built with Psi's own components, with search, markdown twins and a generated `/docs/llms.txt`.

**Architecture:** Astro renders every page to static HTML at build. Psi's React components render inside Astro, and interactive pieces (homepage, docs shell, examples, copy buttons, search) hydrate as islands. Component and token pages are generated from `manifest.json`, `packages/react/docs/<Export>.md` and `resolved/<theme>.json`; guides are MDX in a content collection. Spec: `docs/superpowers/specs/2026-11-16-docs-site-design.md`.

**Tech Stack:** Astro 7.4.12 (Vite 8), `@astrojs/react` 7.1.9, `@astrojs/mdx` 8.1.6, `@astrojs/sitemap` 3.7.5, `@astrojs/check` 0.9.12, Shiki 4.6.0, `marked` 18.2.0, Pagefind 1.6.0, React 19, Vitest 3, Playwright.

## Global Constraints

- **Node 24** (`.nvmrc`). Run `nvm use` in the shell before the first `pnpm` command; check `node -v` prints `v24.x`. Astro 7 needs Node ≥22.23.4. Do not prefix commands with a PATH override.
- **Pin exact versions** (no `^`) for every dependency this plan adds: `astro@7.4.12`, `@astrojs/react@7.1.9`, `@astrojs/mdx@8.1.6`, `@astrojs/sitemap@3.7.5`, `@astrojs/check@0.9.12`, `shiki@4.6.0`, `marked@18.2.0`, `pagefind@1.6.0`. `@astrojs/react` has a peer `oxc-transform-react` (`>=0.145.0 <0.157.0`); install the newest version in that range, pinned.
- **Never hardcode colours** in site CSS or inline styles: bind `var(--psi-*)` (stylelint already lints `apps/**/*.css`). Site-only scale values that the token package does not ship keep the existing `--promo-*` / new `--site-*` custom-property pattern, defined once in `site.css`.
- **Sizes are px numbers** (`24 | 32 | 40 | 48`); **variants are flat** (`accent | accent-subtle | neutral | neutral-subtle | ghost | danger | danger-subtle | outline`). One accent per visual group.
- **No count is typed by hand** in any page, guide or component — counts come from `virtual:psi-facts`.
- **`apps/site` is private** — no changeset for it. If a Psi component in `packages/react` fails to render on the server, fix it in the package with a **patch changeset**, and expect a `vr` refresh from CI if it moves pixels.
- **Do not change Storybook** (`apps/storybook`) — it stays the workbench and the `vr` baseline source.
- **`vr` only passes in CI.** Never run `pnpm vr` locally to update baselines.
- **Decision number D121.** Re-check before opening the PR: no `## Decisions` entry on `origin/main`, and no branch or `.claude/worktrees/*` spec, may own D121. D119 is reserved by another branch; D120 is `d120-field-border`.

- [ ] **Step 4: Make `theme.ts` hydration-safe**

In `apps/site/src/theme.ts`, change the import and the two initial states. The server (and the first client render) use light / no brand; a layout effect corrects to the stored values before paint, so the server HTML and the first client render always match.

```ts
import { useCallback, useEffect, useLayoutEffect, useState } from "react";
```

In `useMode`, replace

```ts
  const [mode, setModeState] = useState<Mode>(() => readStoredMode() ?? systemMode());
```

with

```ts
  // The server has no storage and no OS, so the first render is always
  // light. The boot script has already set <html data-psi-theme>; this only
  // brings React's copy in line, before paint, without a hydration mismatch.
  const [mode, setModeState] = useState<Mode>("light");
  useLayoutEffect(() => {
    setModeState(readStoredMode() ?? systemMode());
  }, []);
```

In `useBrand`, replace

```ts
  const [brand, setBrandState] = useState<BrandVector | null>(
    () => readStoredBrand()?.vector ?? null,
  );
```

with

```ts
  // Same reason as useMode: null on the server and on the first client
  // render, the stored vector before paint.
  const [brand, setBrandState] = useState<BrandVector | null>(null);
  useLayoutEffect(() => {
    setBrandState(readStoredBrand()?.vector ?? null);
  }, []);
```

```ts
import { readFileSync } from "node:fs";
import { createRequire } from "node:module";
import { dirname, join } from "node:path";

const require = createRequire(import.meta.url);
const manifestPath = require.resolve("@handamade/psi-react/manifest.json");
const pkgRoot = join(dirname(manifestPath), "..");

export interface PropInfo {
  name: string;
  type: string;
  required: boolean;
  default: string | number | null;
  description: string;
}
export interface ExportInfo {
  name: string;
  description: string;
  props: PropInfo[];
}

const manifest = JSON.parse(readFileSync(manifestPath, "utf8")) as { components: ExportInfo[] };
const byName = new Map(manifest.components.map((c) => [c.name, c]));

export function allExportNames(): string[] {
  return manifest.components.map((c) => c.name);
}

export function getExport(name: string): ExportInfo {
  const found = byName.get(name);
  if (!found) throw new Error(`No export "${name}" in @handamade/psi-react's manifest.json`);
  return found;
}

/** packages/react/docs/<name>.md, generated by emit-docs.ts. */
export function readGeneratedDoc(name: string): string {
  const raw = readFileSync(join(pkgRoot, "docs", `${name}.md`), "utf8");
  return raw.replace(/^<!--[^\n]*-->\n/, "");
}

export function docSection(md: string, heading: string): string | null {
  const lines = md.split("\n");
  const start = lines.findIndex((l) => l.trim() === `## ${heading}`);
  if (start === -1) return null;
  const rest = lines.slice(start + 1);
  const end = rest.findIndex((l) => l.startsWith("## "));
  return (end === -1 ? rest : rest.slice(0, end)).join("\n").trim();
}
```

```ts
import type { ComponentType } from "react";

const modules = import.meta.glob<{ default: ComponentType }>("./*/*.tsx", { eager: true });
const sources = import.meta.glob<string>("./*/*.tsx", { eager: true, query: "?raw", import: "default" });

export interface ExampleEntry {
  slug: string;
  name: string;
  Component: ComponentType;
  source: string;
}

export const examples: ExampleEntry[] = Object.entries(modules)
  .map(([path, mod]) => {
    const [, slug, file] = path.match(/^\.\/([^/]+)\/([^/]+)\.tsx$/)!;
    return { slug, name: file, Component: mod.default, source: sources[path] };
  })
  .sort((a, b) => `${a.slug}/${a.name}`.localeCompare(`${b.slug}/${b.name}`));

export function examplesFor(slug: string): ExampleEntry[] {
  return examples.filter((e) => e.slug === slug);
}
```

```ts
import { readFileSync } from "node:fs";
import { createRequire } from "node:module";

const require = createRequire(import.meta.url);

export type ThemeName = "light" | "dark" | "acme" | "ember";
export const THEMES: ThemeName[] = ["light", "dark", "acme", "ember"];

interface Oklch {
  l: number;
  c: number;
  h: number;
  alpha?: number;
}
export interface Resolved {
  theme: string;
  colorScheme: string;
  tokens: Record<string, { name: string; oklch: Oklch; hex: string; formula: string; scopes?: string[] }>;
  typography: { name: string; fontSize: number; lineHeight: number; weight: string; cssWeight: number }[];
  display: { name: string; min: number; max: number; vw: number; lineHeight: number; weight: string; tracking: number; cssWeight: number }[];
  scales: {
    space: number[];
    size: number[];
    radius: number[];
    motion: { durations: number[]; easings: Record<string, string> };
    layout: { breakpoints: Record<string, number>; container: Record<string, number>; zIndex: Record<string, number> };
    focusRing: Record<string, number>;
    scopes: Record<string, string[]>;
  };
}

const cache = new Map<ThemeName, Resolved>();
export function resolved(theme: ThemeName = "light"): Resolved {
  if (!cache.has(theme)) {
    const path = require.resolve(`@handamade/psi-tokens/resolved/${theme}.json`);
    cache.set(theme, JSON.parse(readFileSync(path, "utf8")) as Resolved);
  }
  return cache.get(theme)!;
}

export function kebab(name: string): string {
  return name.replace(/([a-z0-9])([A-Z])/g, "$1-$2").toLowerCase();
}

function formatOklch(o: Oklch): string {
  const base = `oklch(${o.l} ${o.c} ${o.h}`;
  return o.alpha != null ? `${base} / ${o.alpha})` : `${base})`;
}

export interface ColorToken {
  name: string;
  cssVar: string;
  formula: string;
  byTheme: Record<ThemeName, { hex: string; oklch: string }>;
}

export function colorTokens(): ColorToken[] {
  return Object.values(resolved("light").tokens).map((t) => ({
    name: t.name,
    cssVar: `--psi-${kebab(t.name)}`,
    formula: t.formula,
    byTheme: Object.fromEntries(
      THEMES.map((th) => {
        const v = resolved(th).tokens[t.name];
        return [th, { hex: v.hex, oklch: formatOklch(v.oklch) }];
      }),
    ) as ColorToken["byTheme"],
  }));
}
```

```ts
import type { ComponentPage } from "../content/component-pages";
import type { TokenPage } from "../content/token-pages";
import { docSection, getExport, readGeneratedDoc } from "./components";
import { THEMES, colorTokens, resolved } from "./tokens";

const fence = (lang: string, code: string) => `\`\`\`${lang}\n${code.trimEnd()}\n\`\`\``;

/** Drops a member doc's "# Name" title (we add "## Name") and its Rules
 * section (identical in every generated doc; the page states it once). */
function memberBody(name: string): string {
  const md = readGeneratedDoc(name);
  const withoutTitle = md.replace(/^# [^\n]*\n/, "");
  const rules = docSection(md, "Rules");
  const withoutRules = rules === null ? withoutTitle : withoutTitle.replace(`## Rules\n\n${rules}`, "");
  return withoutRules.replace(/^## /gm, "### ").trim();
}

export function componentPageMd(page: ComponentPage, examples: { name: string; source: string }[]): string {
  const main = getExport(page.members[0]);
  const rules = docSection(readGeneratedDoc(page.members[0]), "Rules");
  return [
    `# ${page.title}`,
    main.description,
    fence("ts", `import { ${page.members.join(", ")} } from "@handamade/psi-react";`),
    "## Examples",
    ...examples.map((e) => `### ${e.name}\n\n${fence("tsx", e.source)}`),
    ...page.members.map((m) => `## ${m}\n\n${memberBody(m)}`),
    ...(rules ? ["## Rules", rules] : []),
  ].join("\n\n") + "\n";
}

export function tokenPageMd(page: TokenPage): string {
  const r = resolved("light");
  const out: string[] = [`# ${page.title} tokens`];
  switch (page.slug) {
    case "color":
      out.push(
        `| Token | Formula | ${THEMES.join(" | ")} |`,
        `|---|---|${THEMES.map(() => "---").join("|")}|`,
        ...colorTokens().map(
          (t) => `| \`${t.cssVar}\` | \`${t.formula}\` | ${THEMES.map((th) => `${t.byTheme[th].oklch} (${t.byTheme[th].hex})`).join(" | ")} |`,
        ),
      );
      break;
    case "typography":
      out.push("| Token | Size / line | Weight |", "|---|---|---|", ...r.typography.map((t) => `| \`--psi-text-${t.name}\` | ${t.fontSize}/${t.lineHeight} | ${t.weight} (${t.cssWeight}) |`));
      break;
    case "display":
      out.push("| Name | Min–max px | vw | Line height | Weight | Tracking |", "|---|---|---|---|---|---|", ...r.display.map((d) => `| \`${d.name}\` | ${d.min}–${d.max} | ${d.vw} | ${d.lineHeight} | ${d.weight} (${d.cssWeight}) | ${d.tracking} |`));
      break;
    case "spacing":
      out.push(
        `## Space\n\n${r.scales.space.map((v) => `\`--psi-space-${v}\``).join(", ")}`,
        `## Size\n\n${r.scales.size.map((v) => `\`--psi-size-${v}\``).join(", ")}`,
        `## Radius\n\n${r.scales.radius.map((v) => `\`--psi-radius-${v}\``).join(", ")}`,
        `## Focus ring\n\n${Object.entries(r.scales.focusRing).map(([k, v]) => `- \`--psi-focus-ring-${k.replace(/([A-Z])/g, "-$1").toLowerCase()}\`: ${v}px`).join("\n")}`,
      );
      break;
    case "motion":
      out.push(
        `## Durations\n\n${r.scales.motion.durations.map((d) => `\`--psi-duration-${d}\``).join(", ")}`,
        `## Easings\n\n${Object.entries(r.scales.motion.easings).map(([k, v]) => `- \`--psi-ease-${k}\`: \`${v}\``).join("\n")}`,
      );
      break;
    case "layout":
      out.push(
        ...(
          [
            ["Breakpoints", r.scales.layout.breakpoints],
            ["Container", r.scales.layout.container],
            ["Z-index", r.scales.layout.zIndex],
          ] as const
        ).map(([label, values]) => `## ${label}\n\n${Object.entries(values).map(([k, v]) => `- \`${k}\`: ${v}`).join("\n")}`),
      );
      break;
  }
  return out.join("\n\n") + "\n";
}

export function guideMd(entry: { data: { title: string; description: string }; body?: string }): string {
  // MDX import lines and JSX expressions are not markdown; drop import lines,
  // keep everything else as written (guides use fenced code, not live examples).
  const body = (entry.body ?? "").replace(/^import [^\n]*\n/gm, "").trim();
  return `# ${entry.data.title}\n\n${entry.data.description}\n\n${body}\n`;
}

export function overviewMd(): string {
  return [
    "# Psi docs",

    "- Components: /docs/components/button/",
    "- Tokens: /docs/tokens/color/",
    "- MCP: https://psi.kurkin.de/mcp",
  ].join("\n\n") + "\n";
}
```

```tsx
import { useEffect, useRef, useState } from "react";
import { Dialog, Field, Input } from "@handamade/psi-react";

interface Hit {
  url: string;
  title: string;
  excerpt: string;
}
interface Pagefind {
  search: (q: string) => Promise<{ results: { data: () => Promise<{ url: string; meta: { title: string }; excerpt: string }> }[] }>;
}

let pagefind: Promise<Pagefind> | null = null;
function loadPagefind(): Promise<Pagefind> {
  // Loaded on first open only; the path is outside Vite's graph on purpose.
  pagefind ??= import(/* @vite-ignore */ "/pagefind/pagefind.js" as string) as Promise<Pagefind>;
  return pagefind;
}

export function SearchDialog({ open, onClose }: { open: boolean; onClose: () => void }) {
  const [query, setQuery] = useState("");
  const [hits, setHits] = useState<Hit[]>([]);
  const inputRef = useRef<HTMLInputElement>(null);

  useEffect(() => {
    if (!open || query.trim() === "") {
      setHits([]);
      return;
    }
    let cancelled = false;
    loadPagefind()
      .then((pf) => pf.search(query))
      .then((res) => Promise.all(res.results.slice(0, 8).map((r) => r.data())))
      .then((data) => {
        if (!cancelled) setHits(data.map((d) => ({ url: d.url, title: d.meta.title, excerpt: d.excerpt })));
      })
      .catch(() => {
        if (!cancelled) setHits([]);
      });
    return () => {
      cancelled = true;
    };
  }, [open, query]);

  return (
    <Dialog open={open} onClose={onClose} title="Search" width={560} initialFocus={inputRef}>
      <Field label="Search the docs">
        <Input ref={inputRef} type="search" value={query} onChange={(e) => setQuery(e.currentTarget.value)} />
      </Field>
      <ul className="search-results" aria-live="polite">
        {hits.map((h) => (
          <li key={h.url}>
            <a href={h.url}>
              <span className="search-result-title">{h.title}</span>
              {/* Pagefind's excerpt is its own escaped HTML with <mark> highlights. */}
              <span className="search-result-excerpt" dangerouslySetInnerHTML={{ __html: h.excerpt }} />
            </a>
          </li>
        ))}
        {query.trim() !== "" && hits.length === 0 && <li className="search-empty">No results</li>}
      </ul>
    </Dialog>
  );
}
```
