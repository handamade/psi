**Goal:** `Table`'s border follows its rounded corner; a new `TableContainer` scrolls both ways, keeps a sticky header sticking and draws the frame; `numeric` cells do not wrap; a sortable column shows a mark at rest.

**Architecture:** All CSS stays in `packages/react/src/Table/table.module.css`. The sort marks are private inline SVGs in `TableHeaderCell.tsx`.

```css
/* D123: the separate model, not collapse. A collapsed table does not round
   its border: the background takes the curve and the 1px border meets in a
   square corner outside it. --_psi-table-corner is the border's inner radius,
   which the four corner cells take so a sticky header or a hovered first or
   last row stays inside the curve. It is private (not a token): TableContainer
   draws the frame itself and sets it to 0px. */
.table {
  width: 100%;
  border-collapse: separate;
  border-spacing: 0;
  background: var(--psi-table-bg);
  color: var(--psi-table-fg);
  font: var(--psi-text-14-20-regular);
  border: 1px solid var(--psi-table-border);
  border-radius: var(--psi-table-radius);
  --_psi-table-corner: calc(var(--psi-table-radius) - 1px);
}
```

```tsx
type Mark = "both" | "asc" | "desc";

/** The sort mark (D123): private inline SVGs in currentColor, one size in
 * every state, so the slot is always filled and a right-aligned label never
 * shifts when its column becomes sorted. Not exported: the icon set does not
 * change. Same stroke conventions as src/icons. */
function SortMark({ mark }: { mark: Mark }) {
  return (
    <svg width={16} height={16} viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth={2} strokeLinecap="round" strokeLinejoin="round" aria-hidden="true" focusable="false">
      {mark === "both" && (
        <>
          <polyline points="7 9 12 4 17 9" />
          <polyline points="7 15 12 20 17 15" />
        </>
      )}
      {mark === "asc" && (
        <>
          <line x1="12" y1="19" x2="12" y2="5" />
          <polyline points="5 12 12 5 19 12" />
        </>
      )}
      {mark === "desc" && (
        <>
          <line x1="12" y1="5" x2="12" y2="19" />
          <polyline points="19 12 12 19 5 12" />
        </>
      )}
    </svg>
  );
}
```

```tsx
import { useCallback, useId, useLayoutEffect, useRef, useState } from "react";
import type { CSSProperties, HTMLAttributes, ReactNode, Ref } from "react";
import styles from "./table.module.css";
import { TableContainerContext } from "./TableContext.js";

export interface TableContainerProps extends Omit<HTMLAttributes<HTMLDivElement>, "children"> {
  /** Caps the height, so the table scrolls inside and a `stickyHeader` stays
   * at the top. A number is px; a string is any CSS length (`"60vh"`). */
  maxBlockSize?: number | string;
  /** One `Table`. */
  children: ReactNode;
  className?: string;
  /** Forwarded ref to the `<div>`. */
  ref?: Ref<HTMLDivElement>;
}

const NO_NAME_WARNING =
  "Psi TableContainer: no TableCaption inside and no aria-label or aria-labelledby; the scrolling region has no name.";

/** A scrolling region for a wide or long `Table` (D123): scrolls both ways,
 * keeps a `stickyHeader` at its top, and draws the table's frame and corner
 * itself. A tab stop (`role="region"`, `tabIndex={0}`) so the keyboard can
 * scroll it, named by the table's `TableCaption` unless you pass
 * `aria-label` or `aria-labelledby`. Remaining attributes land on the `<div>`. */
export function TableContainer({ maxBlockSize, children, className, style, ref, ...rest }: TableContainerProps) {
  const offeredCaptionId = useId();
  const node = useRef<HTMLDivElement | null>(null);
  const warned = useRef(false);
  const [captionId, setCaptionId] = useState<string | undefined>(undefined);
  const ownName = rest["aria-label"] !== undefined || rest["aria-labelledby"] !== undefined;

  const setRef = useCallback(
    (el: HTMLDivElement | null) => {
      node.current = el;
      if (typeof ref === "function") ref(el);
      else if (ref) (ref as { current: HTMLDivElement | null }).current = el;
    },
    [ref],
  );

  // Every commit: the caption can arrive with data, or change its id. Reading
  // the DOM (not a registration effect) keeps the dev warning free of a
  // first-render false alarm. setCaptionId bails out when nothing changed.
  useLayoutEffect(() => {
    const caption = node.current?.querySelector(":scope > table > caption");
    setCaptionId(caption?.id || undefined);
    if (process.env.NODE_ENV !== "production" && !ownName && !caption && !warned.current) {
      warned.current = true;
      console.warn(NO_NAME_WARNING);
    }
  });

  const sized: CSSProperties | undefined =
    maxBlockSize === undefined
      ? style
      : { ...style, maxBlockSize: typeof maxBlockSize === "number" ? `${maxBlockSize}px` : maxBlockSize };

  return (
    <TableContainerContext.Provider value={offeredCaptionId}>
      <div
        {...rest}
        ref={setRef}
        role="region"
        tabIndex={0}
        aria-labelledby={rest["aria-labelledby"] ?? (rest["aria-label"] !== undefined ? undefined : captionId)}
        className={[styles.container, className].filter(Boolean).join(" ")}
        style={sized}
      >
        {children}
      </div>
    </TableContainerContext.Provider>
  );
}
```

```markdown
---
"@handamade/psi-react": minor
---

A table frame that rounds, and a container that scrolls (D123).

- **`Table`'s border follows its rounded corner** in every theme. The table moves to the separate border model; the corner cells take the inner radius, so a sticky header or a hovered or selected first or last row stays inside the curve. Hover and selected colours now paint the row's cells rather than the `<tr>`: if you styled a row's hover through a `className`, style its cells.
- **`TableContainer`**, new: wraps a `Table` that is wider than its space or must scroll in a fixed height (`maxBlockSize`). It scrolls both ways and keeps a `stickyHeader` at its top, draws the frame itself, and is a named, focusable region (named by the table's `TableCaption`, or by `aria-label`).
- **A `numeric` cell no longer wraps**, so an amount stays on one line.
- **A sortable column shows a two-way mark before it is sorted**; a sorted column shows an up or down arrow in the accent, as before. The marks are SVGs now, not text arrows.
```
